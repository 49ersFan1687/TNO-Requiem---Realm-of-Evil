	custom_effect_tooltip = DRL_southern_urals_reunification_tt
	set_cosmetic_tag = DIR_BEST

	clr_country_flag = is_warlord
	if = {

		for_each_scope_loop = {
			array = global.russian_warlords
			if = {
				limit = {

				}
				every_state = {
					limit = {
						is_core_of = PREV
					}
					PREV = { remove_state_core = PREV }
				}
			}
		}
	}
}
