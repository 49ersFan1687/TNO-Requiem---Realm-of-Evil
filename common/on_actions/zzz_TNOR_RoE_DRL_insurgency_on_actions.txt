on_actions = {
	on_monthly = {
		effect = {
			if = {
				limit = {
					original_tag = DRL
					has_country_flag = DRL_national_insurgency_active
					NOT = { has_country_flag = DRL_national_insurgency_ended }
				}
				set_variable = { DRL_last_monthly_insurgency_growth = 0 }
				random_list = {
					50 = { set_variable = { DRL_last_monthly_insurgency_growth = 0 } }
					10 = { set_variable = { DRL_last_monthly_insurgency_growth = 1 } add_to_variable = { DRL_national_insurgency = 1 } }
					10 = { set_variable = { DRL_last_monthly_insurgency_growth = 2 } add_to_variable = { DRL_national_insurgency = 2 } }
					10 = { set_variable = { DRL_last_monthly_insurgency_growth = 3 } add_to_variable = { DRL_national_insurgency = 3 } }
					10 = { set_variable = { DRL_last_monthly_insurgency_growth = 4 } add_to_variable = { DRL_national_insurgency = 4 } }
					10 = { set_variable = { DRL_last_monthly_insurgency_growth = 5 } add_to_variable = { DRL_national_insurgency = 5 } }
				}
				if = {
					limit = { check_variable = { DRL_national_insurgency > 100 } }
					set_variable = { DRL_national_insurgency = 100 }
				}
			}
		}
	}

	on_daily = {
		effect = {
			if = {
				limit = { original_tag = DRL has_country_flag = DRL_insurgency_operation_running check_variable = { DRL_insurgency_operation_days_left > 0 } }
				subtract_from_variable = { DRL_insurgency_operation_days_left = 1 }
			}
			if = {
				limit = { original_tag = DRL has_country_flag = DRL_national_insurgency_active NOT = { has_country_flag = DRL_national_insurgency_ended } date > 1967.12.31 }
				if = {
					limit = { check_variable = { DRL_national_insurgency > 40 } }
					country_event = { id = DRL_insurgency.11 }
				}
				else = { country_event = { id = DRL_insurgency.10 } }
			}
		}
	}
}
